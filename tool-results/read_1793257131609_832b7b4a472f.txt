    35→function InstantSearch() {
    36→  const { products, stores, go } = useDameen();
    37→  const [q, setQ] = useState('');
    38→  const [open, setOpen] = useState(false);
    39→  const hits = useMemo(() => {
    40→    const k = q.trim();
    41→    if (k.length < 1) return [];
    42→    return products
    43→      .map(p => ({ p, st: stores.find(s => s.id === p.store) }))
    44→      .filter(({ p, st }) => !st?.susp && (p.name.includes(k) || p.cat.includes(k) || (st?.name || '').includes(k)))
    45→      .slice(0, 6);
    46→  }, [q, products, stores]);
    47→  return (
    48→    <div className="relative flex-1 max-w-md">
    49→      <form onSubmit={e => { e.preventDefault(); if (q.trim()) { go({ name: 'search', q: q.trim() }); setOpen(false); } }}>
    50→        <Search size={15} className="absolute start-3 top-1/2 -translate-y-1/2 text-white/50" />
    51→        <Input
    52→          value={q}
    53→          onChange={e => { setQ(e.target.value); setOpen(true); }}
    54→          onFocus={() => setOpen(true)}
    55→          placeholder="ابحث عن منتج أو متجر…"
    56→          className="bg-white/10 border-white/20 text-cream placeholder:text-white/45 rounded-full h-9 ps-9 pe-3 focus-visible:ring-gold/60"
    57→          aria-label="البحث"
    58→        />
    59→      </form>
    60→      {open && q.trim().length > 0 && (
    61→        <>
    62→          <div className="fixed inset-0 z-40" onClick={() => setOpen(false)} />
    63→          <div className="absolute top-full mt-2 inset-x-0 z-50 bg-card text-ink rounded-2xl shadow-2xl border border-border overflow-hidden">
    64→            {hits.length === 0 ? (
    65→              <div className="p-4 text-sm text-muted-foreground text-center">لا نتائج لـ«{q}» — جرّب كلمة أخرى</div>
    66→            ) : hits.map(({ p, st }) => (
    67→              <button
    68→                key={p.id}
    69→                className="w-full flex items-center gap-3 p-2.5 hover:bg-accent text-start transition-colors"
    70→                onClick={() => { go({ name: 'product', id: p.id }); setOpen(false); setQ(''); }}
    71→              >
    72→                {p.imgs[0] ? <img src={p.imgs[0]} alt="" className="w-10 h-10 rounded-lg object-cover" /> : <span className="w-10 h-10 rounded-lg bg-sand flex items-center justify-center">🛍️</span>}
    73→                <span className="flex-1 min-w-0">
    74→                  <span className="block text-sm font-bold truncate">{p.name}</span>
    75→                  <span className="block text-[11px] text-muted-foreground">{st?.name} · {p.cat}</span>
    76→                </span>
    77→                <span className="text-sm font-extrabold text-brand shrink-0">{money(curPrice(p))}</span>
    78→              </button>
    79→            ))}
    80→            <button className="w-full p-2 text-xs font-bold text-brand bg-accent/60 hover:bg-accent" onClick={() => { go({ name: 'search', q: q.trim() }); setOpen(false); }}>
    81→              كل النتائج عن «{q}» ←
    82→            </button>
    83→          </div>
    84→        </>
    85→      )}
    86→    </div>
    87→  );
    88→}
